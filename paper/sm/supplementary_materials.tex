% Crop boxes (content area in bp) of the PDF figures, written by
% figures/scripts/crop_boxes.py; \includecropped[options]{name} includes ../figures/name.pdf
% cropped to its box, or uncropped if there is no box for it.
\IfFileExists{../figures/crop_boxes.csv}{%
	\csvreader{../figures/crop_boxes.csv}%
		{name=\cropname, llx=\cropllx, lly=\croplly, urx=\cropurx, ury=\cropury}{%
		\expandafter\xdef\csname cropbox@\cropname\endcsname{\cropllx\space\croplly\space\cropurx\space\cropury}%
	}%
}{}
\newcommand{\includecropped}[2][]{%
	\ifcsname cropbox@#2\endcsname
		\edef\cropoptions{viewport=\csname cropbox@#2\endcsname, clip, \unexpanded{#1}}%
	\else
		\def\cropoptions{#1}%
	\fi
	\expandafter\includegraphics\expandafter[\cropoptions]{../figures/#2.pdf}%
}
